cmake_minimum_required(VERSION 3.20)
project(onplay_stt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SHERPA_ONNX_ROOT "" CACHE PATH "sherpa-onnx SDK containing include/ and lib/")
find_path(SHERPA_INCLUDE sherpa-onnx/c-api/c-api.h HINTS "${SHERPA_ONNX_ROOT}/include" REQUIRED)
find_library(SHERPA_LIBRARY sherpa-onnx-c-api HINTS "${SHERPA_ONNX_ROOT}/lib" REQUIRED)
add_library(onplay_stt STATIC src/stt.cpp)
set_target_properties(onplay_stt PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(onplay_stt PUBLIC include PRIVATE "${SHERPA_INCLUDE}")
target_link_libraries(onplay_stt PUBLIC "${SHERPA_LIBRARY}")
add_executable(onplay-stt src/main.cpp)
target_include_directories(onplay-stt PRIVATE "${SHERPA_INCLUDE}")
target_link_libraries(onplay-stt PRIVATE onplay_stt)
add_executable(onplay-speakers src/speakers.cpp)
target_include_directories(onplay-speakers PRIVATE "${SHERPA_INCLUDE}")
target_link_libraries(onplay-speakers PRIVATE onplay_stt)
add_executable(onplay-recognize src/recognize.cpp)
target_include_directories(onplay-recognize PRIVATE "${SHERPA_INCLUDE}")
target_link_libraries(onplay-recognize PRIVATE "${SHERPA_LIBRARY}")
